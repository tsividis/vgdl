BasicGame key_handler=Pulse
    SpriteSet

        floor > Immovable color=LIGHTGRAY hidden=True
        avatar > MovingAvatar color=DARKBLUE frameRate=8
        key > Resource color=GREEN limit=1
        lock > Immovable color=RED

        food >
            ingredients > Passive
                boilingwater > img=newset/boilingwater frameRate=8 color=LIGHTBLUE
                rawpasta > img=newset/pasta color=YELLOW
                tomato > img=newset/tomato color=RED
                tuna > img=newset/tuna color=LIGHTRED

            placedIngredients > Passive
                pastainplace > img=newset/pastaplate color=GOLD
                sauceinplace > img=newset/tomatosauce color=ORANGE

        placed > Immovable
            wrongplace  > img=oryx/slash1  
            plateinplace > img=newset/pastasauce 
	wall > Immovable color=GRAY img=newset/floor4


    LevelMapping
        b > boilingwater floor
        p > rawpasta floor
        o > tomato floor
        t > tuna floor
        k > key floor
        l > lock floor
        . > floor
        A > avatar floor

    InteractionSet

        food avatar > bounceForward
        food wall > undoAll 
	food lock > undoAll
	food key > undoAll

        rawpasta boilingwater > transformTo stype=pastainplace scoreChange=4 
	boilingwater rawpasta > killSprite
        rawpasta tuna  > transformTo stype=wrongplace scoreChange=-1 
	tuna rawpasta > killSprite
        rawpasta tomato > transformTo stype=wrongplace scoreChange=-1 
	tomato rawpasta > killSprite
        rawpasta sauceinplace > transformTo stype=wrongplace scoreChange=-1 
	sauceinplace rawpasta > killSprite

        tomato tuna > transformTo stype=sauceinplace scoreChange=4 
	tuna tomato > killSprite
        tomato boilingwater > transformTo stype=wrongplace scoreChange=-1
        boilingwater tomato > killSprite
	tomato pastainplace > transformTo stype=wrongplace scoreChange=-1
        pastainplace tomato > killSprite
	boilingwater sauceinplace > transformTo stype=wrongplace scoreChange=-1 
	sauceinplace boilingwater > killSprite

        tuna boilingwater > transformTo stype=wrongplace scoreChange=-1 
	boilingwater tuna > killSprite
        tuna pastainplace > transformTo stype=wrongplace scoreChange=-1 
	pastainplace tuna > killSprite
        pastainplace sauceinplace > transformTo stype=plateinplace scoreChange=17 
	sauceinplace pastainplace > killSprite

        key avatar > collectResource scoreChange=1
        lock avatar > killIfOtherHasMore resource=key limit=1
        key avatar > killSprite

        avatar wall > stepBack 
        lock avatar > undoAll
