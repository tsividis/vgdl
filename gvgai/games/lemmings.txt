        floor > Immovable color=LIGHTGRAY 
        hole   > Immovable color=LIGHTBLUE 
        shovel > Flicker color=BROWN limit=1 singleton=True 

        entrance > SpawnPoint total=20 spawnCooldown=50 stype=lemming color=PURPLE 
        goal > Immovable color=GREEN #This is actually the goal, but Chaser only works when stype=goal for some reason

        avatar  > ShootAvatar stype=shovel color=DARKBLUE
        lemming > Chaser  stype=goal speed=0.1 color=RED

        wall > Immovable color=GRAY 
    LevelMapping
        x > floor goal
        e > floor entrance
        h > floor hole
        . > floor
        A > floor avatar
        w > floor wall

        MultiSpriteCounter stype1=entrance stype2=lemming limit=0 win=True

        lemming wall > stepBack
