BasicGame
  SpriteSet
    floor > Immovable color=GRAY
    city  > Immovable color=PURPLE
    bullet > Flicker limit=5 color=GOLD shrinkfactor=0.7
    movable >
        avatar > ShootAvatar stype=bullet color=DARKBLUE
        goalObjects >
            enemy > Chaser stype=city
                enemy_slow  > color=GREEN speed=0.1
                enemy_fast  > color=RED speed=0.3
            niceGuy > Chaser stype=city color=ORANGE speed=0.1
    death   > Immovable
    portal  > SpawnPoint
        enemySlowPortal > stype=enemy_slow   cooldown=100  total=2 
        enemyFastPortal > stype=enemy_fast   cooldown=100  total=2 
        niceGuyPortal   > stype=niceGuy      cooldown=100  total=2

    wall > Immovable color=BLACK

  LevelMapping
    c > city floor
    m > enemySlowPortal floor
    f > enemyFastPortal floor
    n > niceGuyPortal floor
    . > floor
    A > avatar floor

  InteractionSet
    movable EOS    > stepBack
    movable wall   > stepBack
    enemy city     > killSprite
    city enemy     > killSprite
    enemy city     > changeScore value=-1
    niceGuy bullet > transformTo stype=death
    niceGuy bullet > changeScore value=-1
    enemy bullet   > killSprite
    enemy bullet   > changeScore value=2
    niceGuy city   > killSprite
    niceGuy city   > changeScore value=2

  TerminationSet
    SpriteCounter stype=city   win=False
    MultiSpriteCounter stype1=goalObjects stype2=portal limit=0 win=True
    MultiSpriteCounter stype1=death limit=1 win=False
    MultiSpriteCounter stype1=death limit=2 win=False
    MultiSpriteCounter stype1=death limit=3 win=False
    MultiSpriteCounter stype1=death limit=4 win=False
