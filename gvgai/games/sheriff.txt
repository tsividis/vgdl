BasicGame
    SpriteSet

        floor > Immovable color=LIGHTGRAY img=oryx/backBiege
        base    > Immovable    color=WHITE img=oryx/wall4
        avatar  > ShootAvatar color=BLACK stype=bullet img=newset/sheriff1
        surround > Immovable    color=YELLOW img=dirt invisible=True
        bullet > Missile color=BLUE img=oryx/orb3 singleton=True

        turn > Immovable color=BROWN invisible=True hidden=True
            turnup > orientation=UP img=dirt
            turndown > orientation=DOWN img=dirt
            turnleft > orientation=LEFT img=dirt
            turnright > orientation=RIGHT img=dirt

        bandit > Bomber color=RED prob=0.1 cooldown=6 speed=1 img=newset/bandit1
            bup > orientation=UP spawnorientation=LEFT stype=missile
            bdown > orientation=DOWN spawnorientation=RIGHT stype=missile
            bleft > orientation=LEFT spawnorientation=DOWN stype=missile
            bright > orientation=RIGHT spawnorientation=UP stype=missile

        missile > Missile color=GREEN img=oryx/orb1 speed=0.6

        wall > Immovable color=GRAY autotiling=True img=oryx/dirtwall

    LevelMapping
        0 > floor base
        u > floor surround bup
        d > floor surround bdown
        l > floor surround bleft
        r > floor surround bright
        s > floor surround
        A > floor avatar
        1 > floor turnup
        2 > floor turnleft
        3 > floor turndown
        4 > floor turnright
        w > floor wall
        . > floor

    TerminationSet
        SpriteCounter      stype=avatar             limit=0 win=False
        SpriteCounter      stype=bandit             limit=0 win=True
        Timeout timeout=1000 win=True
        
    InteractionSet
        avatar surround > stepBack
	avatar base > stepBack 
	avatar wall > stepBack

        base missile > killSprite
	missile base > killSprite
	base bullet > killSprite
	bullet base > killSprite
        EOS missile > killSprite 
	EOS bullet > killSprite

        bleft turndown > transformTo stype=bdown
        bright turnup > transformTo stype=bup
        bdown turnright > transformTo stype=bright
        bup turnleft > transformTo stype=bleft

        bandit bullet > killSprite 
	bullet bandit > killSprite 
	bullet bandit > changeScore value=1
        avatar missile > killSprite 
	missile avatar > killSprite 
	missile avatar > changeScore value=-1

        bullet wall > killSprite
