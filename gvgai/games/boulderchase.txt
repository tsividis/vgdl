BasicGame

		background > Immovable color=LIGHTGRAY img=oryx/backBlack hidden=True
		wall > Immovable color=GRAY autotiling=true img=oryx/dirtWall
		sword > Flicker color=YELLOW limit=1 singleton=True img=oryx/pickaxe
		dirt > Immovable color=BROWN img=oryx/backLBrown
		exitdoor > Immovable color=GREEN img=oryx/door2
		diamond > Resource color=GOLD limit=10 img=oryx/diamond3
		boulder > Missile orientation=DOWN color=GRAY speed=0.2 img=oryx/mineral1

                avatar  > ShootAvatar  color=WHITE stype=sword frameRate=8 img=oryx/spelunky
                crab > Chaser stype=avatar speed=0.1 color=RED img=oryx/scorpion2
                butterfly > Chaser stype=avatar speed=0.1 color=BLUE img=oryx/bat2

		. > background dirt
		e > background exitdoor
		o > background boulder
		x > background diamond
		c > background crab
		b > background butterfly
		- > background
		A > background avatar

	InteractionSet
                dirt avatar > killSprite 
                dirt sword > killSprite
                diamond avatar > collectResource 
                diamond avatar > changeScore value=2

                avatar wall > stepBack 
                avatar boulder > stepBack
                crab wall > stepBack 
                crab boulder > stepBack
                butterfly wall > stepBack 
                butterfly boulder > stepBack

                avatar boulder > killIfFromAbove 
                avatar boulder > changeScore value=-1
                avatar butterfly > killSprite 
                avatar butterfly > changeScore value=-1 
                avatar crab > killSprite 
                avatar crab > changeScore value=-1 

                enemy boulder > killIfFromAbove 
                enemy boulder > changeScore value=5 

		boulder dirt > stepBack 
		boulder wall > stepBack
		boulder diamond > stepBack
		boulder boulder > stepBack

		dirt butterfly > killSprite
		dirt crab > transformTo stype=diamond
		crab butterfly > killSprite
		butterfly crab > transformTo stype=diamond 
                butterfly crab > changeScore value=1
		exitdoor avatar > killIfOtherHasMore resource=diamond limit=9
