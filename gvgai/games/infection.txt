BasicGame
    SpriteSet
        floor > Immovable color=LIGHTGRAY img=oryx/backGrey hidden=True
        sword > Flicker color=BLACK limit=5 singleton=True img=oryx/sword1
        entrance > SpawnPoint total=5 color=PINK cooldown=100 stype=doctor img=oryx/dooropen1 portal=True
        virus > Immovable color=RED img=oryx/slime4
        moving >
            avatar > ShootAvatar stype=sword
                 normal   > color=WHITE img=newset/girl2
                 carrier  > color=BROWN img=newset/girl2_c frameRate=8
            npc    > RandomNPC speed=0.25  cons=1
                 host     > color=GREEN img=newset/man3
                 infected > color=ORANGE speed=0.5 img=newset/cursedman frameRate=8
                 doctor > color=BLUE speed=0.2 img=newset/nurse
        wall > Immovable color=GRAY img=oryx/wall3 autotiling=True

    TerminationSet
        SpriteCounter stype=host   win=True

    InteractionSet
        doctor sword    > killSprite 
	doctor sword  > changeScore value=2
        moving wall       > stepBack
        carrier doctor  > transformTo stype=normal  
	carrier doctor > changeScore value=-1
        host carrier      > transformTo stype=infected 
	host carrier > changeScore value=2
        infected doctor > transformTo stype=host
        normal infected > transformTo stype=carrier 
	normal virus  > transformTo stype=carrier
        host infected > transformTo stype=infected 
	host virus    > transformTo stype=infected
        doctor virus    > killSprite

    LevelMapping
        1 > floor doctor
        0 > floor host
        x > floor virus
        A > floor normal
        e > floor entrance
        . > floor
